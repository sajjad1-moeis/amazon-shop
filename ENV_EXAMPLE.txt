# ============================================
# Amazon Shop Next.js Frontend - Environment Variables Example
# ============================================
# این فایل را به .env کپی کنید و مقادیر را تنظیم کنید
# دستور: cp ENV_EXAMPLE.txt .env

# ============================================
# API Configuration
# ============================================
# آدرس پایه Backend API (سفارش، سبد، پرداخت، ذخیره محصول)
# Production: http://107.161.175.45:8080/api
NEXT_PUBLIC_API_URL=http://107.161.175.45:8080/api

# آدرس اسکرپر پایتون (جستجوی مستقیم فرانت → پایتون). بدون /api در انتها.
# مثال: http://107.161.175.45:5000
NEXT_PUBLIC_SCRAPER_URL=http://107.161.175.45:5000

# آدرس پایه سایت
# Production (IP): http://107.161.175.45
NEXT_PUBLIC_BASE_URL=http://107.161.175.45

# ============================================
# Next.js Configuration
# ============================================
# Environment: production | development
NODE_ENV=production

# Disable Next.js telemetry
NEXT_TELEMETRY_DISABLED=1

# ============================================
# Docker Configuration (Optional)
# ============================================
# Port for Next.js server (default: 3000)
PORT=3000

# Hostname for Next.js server (default: 0.0.0.0)
HOSTNAME=0.0.0.0

# ============================================
# Additional Configuration (Optional)
# ============================================
# Enable bundle analyzer (true | false)
# ANALYZE=false

# ============================================
# Notes
# ============================================
# 1. تمام متغیرهای NEXT_PUBLIC_* در client-side در دسترس هستند
# 2. برای تغییرات در environment variables، container را restart کنید
# 3. در production، حتماً NODE_ENV=production تنظیم کنید
# 4. NEXT_PUBLIC_BASE_URL باید با آدرس Backend API شما مطابقت داشته باشد
